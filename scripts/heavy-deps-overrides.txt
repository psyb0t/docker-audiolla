# Override tensorflow out of the resolution graph.
# basic-pitch on Linux+Python≥3.11 declares tensorflow as a base dep, but
# the library correctly falls back to its bundled ONNX model when only
# onnxruntime is installed (TF_PRESENT=False → ONNX backend selected).
# Excluding tensorflow keeps the image lean and avoids the ≥500 MB overhead.
tensorflow ; sys_platform == "never"
tensorflow-cpu ; sys_platform == "never"

# Force numpy 2.1.3 to resolve the audio-separator vs deepfilternet conflict:
#   audio-separator==0.44.2 requires numpy>=2
#   deepfilternet==0.5.6 declares numpy<2.0 (overly conservative; works fine with 2.x)
# uv overrides ignore all other constraints for the overridden package, so
# numpy==2.1.3 is installed regardless of either package's declared bounds.
numpy==2.1.3
